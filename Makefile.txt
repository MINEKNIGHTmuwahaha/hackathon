.PHONY: setup data test smoke cli-all clean

setup:
	python -m pip install -r requirements.txt pyyaml pytest fpdf2

data:
	python -m src.cli build-dataset

test:
	python -m pytest -q

smoke:
	python -m src.smoke

cli-all:
	python -m src.cli validate
	python -m src.cli build-dataset
	python -m src.cli audit
	python -m src.cli split
	python -m src.cli visualize

clean:
	rmdir /s /q .pytest_cache 2>nul
	rmdir /s /q src\__pycache__ 2>nul
	rmdir /s /q tests\__pycache__ 2>nul
